§3 — الحكمُ على الدعوى: صحّت في شطرٍ وسقطت في شطر
=========================================================================
  **الشطرُ المحميّ (P1 · P2)** — البابانِ وحيدانِ في مسلكِ القراءة: مخرجاتُ السجلِّ
  كلُّها تمرّ بموضعٍ واحدٍ يَنسخ ويُجمّد (`sealed`، من علاجِ T-168)، فلمسُ العقدِ
  المُعادِ يرمي `TypeError` والحالةُ لا تتغيّر. وهذا الشطرُ **قُيِّد حكمًا في الشجرة**
  (§4).

  **الشطرُ المخروق (P3 · P4b · P5 · P6)** — و`importAll` بابٌ ثالثٌ لم يُحسب:
  لا يفحص إلّا **البنية** (مصفوفةٌ · تسلسلٌ صحيح · لا تكرار · بادئةُ المعرّف ·
  نوعٌ معلن · قيمةٌ عددٌ موجب · `endsAt` عدد)، فيمرّ منه ما يرفضه البابان:

    (1) **قيمةٌ فوق سقفِ النوع**: 5,000,000 ⇒ **900,000,000** — و`committedValue`
        الذي تقوم عليه حراسةُ سقفِ النظير **تبعها** إلى 900,000,000.
    (2) **بعثُ مُنهًى**: عقدٌ أُنهي ببابِه الوحيدِ عاد **قائمًا** بتصفيرِ حقولِ
        الإنهاءِ الثلاثةِ في اللقطة — والحقولُ صارت `readonly` بـT-168، لكنّ
        `readonly` حكمُ مُصنِّفٍ لا حكمُ تشغيل، ومسلكُ الاستعادةِ يبني كائنًا جديدًا.
    (3) **تجاوزُ سقفِ النظير (ثابت 7)**: السقفُ **3** ودخل **50** عقدًا قائمًا لنظيرٍ
        واحد.
    (4) **تجاوزُ سقفِ المدّة (ثابت 5)**: السقفُ **60 يومًا** ودخل عقدٌ قائمٌ إلى
        **3650 يومًا**.

  ولا واحدةٌ من الأربعِ رفعت مخالفةً ولا كتبت في التدقيقِ إلّا سطرًا واحدًا محايدًا
  (`external_contracts_restored`) بعددِ العقودِ لا بحكمٍ عليها.

  وهذا **نظيرُ T-60 حرفًا**: هناك كان البابُ الخلفيُّ `treasury.disburse` يتجاوز
  سقفَ الدفعِ، وهنا `importAll` يتجاوز سقفَ العقد. والدرسُ واحد: **حارسٌ على بابٍ
  واحدٍ ليس حارسًا على الثابت**.

§6 — المسبارُ نصًّا (لإعادةِ الإنتاج: يُوضع في tests/external/ ثمّ يُبنى ويُشغَّل)

import test from 'node:test';
import assert from 'node:assert/strict';
import { COMPUTE, DAY, KIND_COMPUTE, assessRisk, contractState, signContract, terminateContract } from './helpers';

test('PROBE: مسارات حقن الانحراف على عقد قائم', () => {
  const h = contractState();
  assessRisk(h);
  const s = signContract(h, KIND_COMPUTE, { value: 5_000_000, days: 30 });
  assert.equal(s.ok, true);
  if (!s.ok) return;
  const id = s.contract.contractId;
  const reg = h.s.contracts;

  // 1) get() ثم الكتابة
  const got = reg.get(id) as any;
  let e1 = 'قُبل';
  try { got.value = 999_999_999; } catch (err) { e1 = (err as Error).constructor.name; }
  console.log('P1 get().value =', e1, '· القيمة الآن', reg.get(id)?.value);

  // 2) list() الكتابة في المصفوفة
  const arr = reg.list() as any;
  let e2 = 'قُبل';
  try { arr.push({}); } catch (err) { e2 = (err as Error).constructor.name; }
  console.log('P2 list().push =', e2, '· العدد', reg.list().length);

  // 3) exportAll ثم تعديل ثم importAll — رفع القيمة فوق سقف النوع
  const snap = reg.exportAll();
  const bumped = { contracts: snap.contracts.map((c: any) => ({ ...c, value: 900_000_000 })), sequence: snap.sequence };
  let e3 = 'قُبل';
  try { reg.importAll(bumped as any, 'authority:crown_guard'); } catch (err) { e3 = (err as Error).message; }
  console.log('P3 importAll(قيمة 900م) =', e3, '· القيمة الآن', reg.get(id)?.value, '· الملتزَم', reg.committedValue(COMPUTE, h.now));

  // 4) بعث عقد مُنهًى عبر importAll
  const h2 = contractState();
  assessRisk(h2);
  const s2 = signContract(h2, KIND_COMPUTE, { value: 1_000_000, days: 30 });
  if (!s2.ok) return;
  const t = terminateContract(h2, s2.contract.contractId);
  console.log('P4a إنهاء =', t.ok, '· قائم؟', h2.s.contracts.isActive(s2.contract.contractId, h2.now));
  const snap2 = h2.s.contracts.exportAll();
  const revived = { contracts: snap2.contracts.map((c: any) => ({ ...c, terminatedAt: null, terminatedBy: null, terminationReason: null })), sequence: snap2.sequence };
  let e4 = 'قُبل';
  try { h2.s.contracts.importAll(revived as any, 'authority:crown_guard'); } catch (err) { e4 = (err as Error).message; }
  console.log('P4b بعثُ مُنهًى =', e4, '· قائم؟', h2.s.contracts.isActive(s2.contract.contractId, h2.now));

  // 5) تجاوز سقف عدد العقود لنظير عبر importAll
  const h3 = contractState();
  assessRisk(h3);
  const s3 = signContract(h3, KIND_COMPUTE, { value: 1_000_000, days: 30 });
  if (!s3.ok) return;
  const snap3 = h3.s.contracts.exportAll();
  const many = { contracts: Array.from({ length: 50 }, (_, i) => ({ ...(snap3.contracts[0] as any), contractId: `extc:inj-${i}` })), sequence: snap3.sequence + 50 };
  let e5 = 'قُبل';
  try { h3.s.contracts.importAll(many as any, 'authority:crown_guard'); } catch (err) { e5 = (err as Error).message; }
  console.log('P5 50 عقدًا لنظير واحد =', e5, '· القائم', h3.s.contracts.active(h3.now).length, '· السقف', h3.s.contracts.report(h3.now).maxOpenPerCounterparty, '· الملتزَم', h3.s.contracts.committedValue(COMPUTE, h3.now));

  // 6) مدة تتجاوز maxTermMs عبر importAll
  const h4 = contractState();
  assessRisk(h4);
  const s4 = signContract(h4, KIND_COMPUTE, { value: 1_000_000, days: 30 });
  if (!s4.ok) return;
  const snap4 = h4.s.contracts.exportAll();
  const longer = { contracts: [{ ...(snap4.contracts[0] as any), endsAt: h4.now + 3650 * DAY }], sequence: snap4.sequence };
  let e6 = 'قُبل';
  try { h4.s.contracts.importAll(longer as any, 'authority:crown_guard'); } catch (err) { e6 = (err as Error).message; }
  console.log('P6 مدة 10 سنوات =', e6, '· السقف يومًا', h4.s.contracts.report(h4.now).maxTermMs / DAY, '· قائم؟', h4.s.contracts.isActive(s4.contract.contractId, h4.now + 3000 * DAY));
});

test('**حقنُ انحرافٍ: لمسُ مخرجِ السجلِّ يُرمى ولا يُغيّر الحالة** (T-68 · نظير T-168)', () => {
  const h = contractState();
  assert.equal(assessRisk(h).ok, true);
  const signed = signContract(h, KIND_COMPUTE, { value: 5_000_000, days: 30 });
  assert.equal(signed.ok, true);
  if (!signed.ok) return;
  const id = signed.contract.contractId;

  const got = h.s.contracts.get(id) as unknown as { value: number };
  assert.throws(() => { got.value = 900_000_000; }, TypeError, 'المخرجُ مجمَّدٌ فيرمي');
  assert.equal(h.s.contracts.get(id)?.value, 5_000_000, 'والحالةُ لم تتغيّر');

  const listed = h.s.contracts.list() as unknown as unknown[];
  assert.throws(() => { listed.push({}); }, TypeError, 'والقائمةُ مجمَّدةٌ كذلك');
  assert.equal(h.s.contracts.list().length, 1);
});

test('**حقنُ انحرافٍ: `importAll` لا يقبل قيمةً تتجاوز سقفَ النوع** — شرط قبول T-68', () => {
  const h = contractState();
  assert.equal(assessRisk(h).ok, true);
  const signed = signContract(h, KIND_COMPUTE, { value: 5_000_000, days: 30 });
  assert.equal(signed.ok, true);
  if (!signed.ok) return;

  const snapshot = h.s.contracts.exportAll();
  const inflated = {
    contracts: snapshot.contracts.map((c) => ({ ...c, value: 900_000_000 })),
    sequence: snapshot.sequence,
  };
  assert.throws(
    () => h.s.contracts.importAll(inflated, CROWN_GUARD),
    /RECOVERY_CONTRACT_INVALID/,
    'قيمةٌ يرفضها `sign` لا تدخل من مسلكِ الاستعادة',
  );
  assert.equal(h.s.contracts.get(signed.contract.contractId)?.value, 5_000_000);
  assert.equal(h.s.contracts.committedValue(COMPUTE, h.now), 5_000_000, 'ولا يتبعُها الملتزَم');
});

test('**حقنُ انحرافٍ: `importAll` لا يبعث عقدًا مُنهًى** — شرط قبول T-68', () => {
  const h = contractState();
  assert.equal(assessRisk(h).ok, true);
  const signed = signContract(h, KIND_COMPUTE, { value: 1_000_000, days: 30 });
  assert.equal(signed.ok, true);
  if (!signed.ok) return;
  const id = signed.contract.contractId;
  assert.equal(terminateContract(h, id).ok, true);
  assert.equal(h.s.contracts.isActive(id, h.now), false);

  const snapshot = h.s.contracts.exportAll();
  const revived = {
    contracts: snapshot.contracts.map((c) => ({
      ...c, terminatedAt: null, terminatedBy: null, terminationReason: null,
    })),
    sequence: snapshot.sequence,
  };
  assert.throws(
    () => h.s.contracts.importAll(revived, CROWN_GUARD),
    /RECOVERY_CONTRACT_INVALID/,
    'الإنهاءُ بابٌ واحدٌ، ونقضُه ليس مسلكَ استعادة',
  );
  assert.equal(h.s.contracts.isActive(id, h.now), false, 'ويبقى مُنهًى');
});

test('**حقنُ انحرافٍ: `importAll` لا يتجاوز سقفَ عددِ العقودِ لنظيرٍ ولا سقفَ المدّة** — شرط قبول T-68', () => {
  const h = contractState();
  assert.equal(assessRisk(h).ok, true);
  const signed = signContract(h, KIND_COMPUTE, { value: 1_000_000, days: 30 });
  assert.equal(signed.ok, true);
  if (!signed.ok) return;
  const snapshot = h.s.contracts.exportAll();
  const first = snapshot.contracts[0];
  assert.ok(first);

  const ceiling = h.s.contracts.report(h.now).maxOpenPerCounterparty;
  const flooded = {
    contracts: Array.from({ length: ceiling + 10 }, (_, i) => ({ ...first, contractId: `extc:injected-${i}` })),
    sequence: snapshot.sequence + ceiling + 10,
  };
  assert.throws(
    () => h.s.contracts.importAll(flooded, CROWN_GUARD),
    /RECOVERY_CONTRACT_INVALID/,
    'سقفُ النظيرِ (ثابت 7) يُحكم على المستعادِ كما يُحكم على الموقَّع',
  );
  assert.equal(h.s.contracts.active(h.now).length, 1);

  const overlong = {
    contracts: [{ ...first, endsAt: h.now + 3650 * DAY }],
    sequence: snapshot.sequence,
  };
  assert.throws(
    () => h.s.contracts.importAll(overlong, CROWN_GUARD),
    /RECOVERY_CONTRACT_INVALID/,
    'ومدّةٌ فوق `maxTermMs` (ثابت 5) لا تدخل من الخلف',
  );
  assert.equal(h.s.contracts.get(signed.contract.contractId)?.endsAt, h.now + 30 * DAY);
});
